cmake_minimum_required(VERSION 3.22.1)
project(llama_probe LANGUAGES C CXX)

if(NOT DEFINED LLAMA_CPP_ROOT)
    message(FATAL_ERROR "请通过 -DLLAMA_CPP_ROOT 指定 llama.cpp 源码目录")
endif()
if(NOT DEFINED PROBE_BACKEND)
    set(PROBE_BACKEND "CPU")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
if(PROBE_BACKEND STREQUAL "VULKAN")
    set(LLAMA_VULKAN ON CACHE BOOL "" FORCE)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    set(PROBE_LIBRARY llama_probe_vulkan)
else()
    set(LLAMA_VULKAN OFF CACHE BOOL "" FORCE)
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
    set(PROBE_LIBRARY llama_probe_cpu)
endif()

if(DEFINED VULKAN_HPP_ROOT)
    include_directories(${VULKAN_HPP_ROOT})
endif()
if(DEFINED SPIRV_HEADERS_ROOT)
    include_directories(${SPIRV_HEADERS_ROOT}/include)
endif()
add_subdirectory(${LLAMA_CPP_ROOT} ${CMAKE_BINARY_DIR}/llama.cpp)
add_library(${PROBE_LIBRARY} SHARED llama_probe.cpp)
target_include_directories(${PROBE_LIBRARY} PRIVATE
    ${LLAMA_CPP_ROOT}/include
    ${LLAMA_CPP_ROOT}/ggml/include
)
target_compile_features(${PROBE_LIBRARY} PRIVATE cxx_std_17)
target_compile_options(${PROBE_LIBRARY} PRIVATE -O3 -fexceptions -frtti)
if(PROBE_BACKEND STREQUAL "VULKAN")
    target_compile_definitions(${PROBE_LIBRARY} PRIVATE PROBE_VULKAN=1)
endif()
target_link_libraries(${PROBE_LIBRARY} PRIVATE llama ggml android log)
if(TARGET ggml-vulkan)
    target_link_libraries(${PROBE_LIBRARY} PRIVATE ggml-vulkan)
endif()
if(PROBE_BACKEND STREQUAL "VULKAN")
    find_package(Vulkan COMPONENTS glslc REQUIRED)
    target_link_libraries(${PROBE_LIBRARY} PRIVATE Vulkan::Vulkan)
endif()
